#!/usr/bin/env python3
"""
A288957: Number of spanning trees in the n X n king graph.
Exact Matrix-Tree theorem: delete one row/col from the Laplacian, take the
determinant via fraction-free Gaussian elimination (Bareiss algorithm) --
all-integer, no rounding.

King graph on n x n grid: vertex (r,c), r,c in 0..n-1. Edges to all up-to-8
king-neighbours (orthogonal + diagonal) within bounds.
"""
import sys, time, resource
sys.path.insert(0, "/Users/june/.sanshar/problems/research/SCOUT_G")
from bareiss import bareiss_det

def king_laplacian(n):
    N = n*n
    def idx(r,c): return r*n+c
    deg = [0]*N
    adj = [[] for _ in range(N)]
    for r in range(n):
        for c in range(n):
            i = idx(r,c)
            for dr in (-1,0,1):
                for dc in (-1,0,1):
                    if dr==0 and dc==0: continue
                    rr,cc = r+dr, c+dc
                    if 0<=rr<n and 0<=cc<n:
                        j = idx(rr,cc)
                        adj[i].append(j)
    L = [[0]*(N-1) for _ in range(N-1)]  # delete last vertex (row/col N-1)
    for i in range(N-1):
        L[i][i] = len(adj[i])
        for j in adj[i]:
            if j < N-1:
                L[i][j] -= 1
    return L

def spanning_trees_king(n):
    L = king_laplacian(n)
    return bareiss_det(L)
