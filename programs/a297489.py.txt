#!/usr/bin/env python3
"""
A297489: number of MAXIMAL matchings in the n x n KING graph.

King graph on the n x n grid: vertex (r,c), r,c in 0..n-1. Edges connect a cell
to all 8 king-neighbours: orthogonal (r,c)-(r+-1,c),(r,c)-(r,c+-1) AND the two
diagonals (r,c)-(r+-1,c+-1).

A matching = set of pairwise vertex-disjoint edges. MAXIMAL = no edge can be
added, i.e. the set of UNMATCHED vertices is an INDEPENDENT set in the king graph
(no two king-adjacent vertices are both unmatched).

METHOD (exact, deterministic column transfer over big-ints; M25 KING analog):
  Process column by column, left to right. Between column c and c+1 the coupling
  edges are: horizontal (r,c)-(r,c+1) and the two diagonals (r,c)-(r-1,c+1),
  (r,c)-(r+1,c+1). So a column-c vertex (r,c) can be matched by a CROSS edge into
  next-column rows {r-1, r, r+1}; a next-column vertex (r',c+1) can be reached from
  column-c rows {r'-1, r', r'+1}. Matching => at most one cross edge per vertex.

  Per-row OUTGOING state (of next column) takes 3 values:
     0 = "clear": (r',c+1) not pre-matched and not forced -> free to be matched
         or unmatched when column c+1 is processed.
     1 = "locked": a cross edge from column c matches (r',c+1) -> it is matched and
         may take no further edge.
     2 = "must-match": some column-c neighbour {r'-1,r',r'+1} is UNMATCHED, so for
         maximality (r',c+1) MUST be matched (by a vertical edge inside c+1 or a
         cross edge to c+2); it is not yet matched and can still take an edge.
  The full boundary state = tuple of these over all n rows (packed 2 bits/row).

  Incoming state a[r] for (r,c): 1=locked (no more edges), 2=must-match (forced),
  0=free (and implies NO left-column neighbour is unmatched, so (r,c) may itself be
  unmatched). Column 0 incoming = all 0 (phantom matched left column).

  Column transition (inner recursion over rows r=0..n-1), per vertex decide:
    - ve_down: vertical edge (r,c)-(r+1,c)  (carried as prev_v into row r+1)
    - cross edges (up,straight,down) from (r,c) to next rows r-1,r,r+1; sum <= 1,
      up only if r>0, down only if r<n-1, all 0 in the LAST column.
    deg(r) = [a[r]==1] + prev_v + ve_down + up + straight + down  (matching: <=1)
    a[r]==1 -> locked: prev_v=ve_down=up=straight=down=0.
    a[r]==2 -> must be matched: deg==1.
    a[r]==0 -> free: matched (deg==1) or UNMATCHED (deg==0).
    unmatched(r) forbidden if a[r]!=0 or prev_un (two vertically-adjacent unmatched).
    An unmatched (r,c) taints next rows {r-1,r,r+1} -> their outgoing state becomes
    2 (must-match) unless locked(1). Cross-edge reservations R[next-row] enforced <=1.

  Sliding-window carries (entering row r): prev_v, prev_un, pA,pB (partial R of
  next rows r-1,r), fA,fB (partial force of next rows r-1,r). Next-row r-1 is
  finalized while processing row r (all its sources r-2,r-1,r are then known);
  next-row n-1 finalized in a virtual row n.

  LAST column (c=n-1): no cross edges; just count valid fillings (unmatched
  last-column vertices are fine, no right neighbour to force).
"""
import sys, time, resource

# ---- state packed: 2 bits/row, digit in {0,1,2}; 0=clear 1=locked 2=must-match ----

def cross_options(r, n, last):
    if last:
        return ((0, 0, 0),)
    opts = [(0, 0, 0), (0, 1, 0)]        # none ; straight
    if r > 0:
        opts.append((1, 0, 0))           # up  -> next row r-1
    if r < n - 1:
        opts.append((0, 0, 1))           # down-> next row r+1
    return opts


def column_transitions(n, a_packed, last):
    a = [(a_packed >> (2 * r)) & 3 for r in range(n)]
    out = {}
    ob = [0] * n

    def rec(r, prev_v, prev_un, pA, pB, fA, fB):
        if r == n:
            # finalize next-row n-1: R = pA, force = fA
            if pA > 1:
                return
            ob[n - 1] = 1 if pA == 1 else (2 if fA else 0)
            key = 0
            for i in range(n - 1, -1, -1):
                key = (key << 2) | ob[i]
            out[key] = out.get(key, 0) + 1
            return
        ar = a[r]
        for ve_down in (0, 1):
            if ve_down and (r == n - 1 or prev_v):
                continue
            for up, straight, down in cross_options(r, n, last):
                if ar == 1:
                    if prev_v or ve_down or up or straight or down:
                        continue
                    matched = True
                else:
                    deg = prev_v + ve_down + up + straight + down
                    if deg > 1:
                        continue
                    matched = deg == 1
                    if ar == 2 and not matched:
                        continue
                unmatched_r = not matched
                if unmatched_r and (ar != 0 or prev_un):
                    continue
                Rprev = pA + up
                if Rprev > 1:
                    continue
                if r - 1 >= 0:
                    ob[r - 1] = 1 if Rprev == 1 else (2 if (fA or unmatched_r) else 0)
                rec(r + 1, ve_down, unmatched_r,
                    pB + straight, down,
                    fB or unmatched_r, unmatched_r)

    rec(0, 0, False, 0, 0, False, False)
    return out


def a_transfer(n, verbose=False, cache_trans=True):
    if n == 0:
        return 1
    if n == 1:
        return 1          # single vertex, no edges: empty matching is maximal
    dist = {0: 1}
    cache = {} if cache_trans else None   # None => recompute each column (memory-light)
    for c in range(n):
        last = (c == n - 1)
        nd = {}
        ndget = nd.get
        for st, cnt in dist.items():
            if last:
                tr = column_transitions(n, st, True)
            elif cache is not None:
                tr = cache.get(st)
                if tr is None:
                    tr = column_transitions(n, st, False)
                    cache[st] = tr
            else:
                tr = column_transitions(n, st, False)
            for o, m in tr.items():
                nd[o] = ndget(o, 0) + cnt * m
        dist = nd
        if verbose:
            rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024 * 1024)
            csz = len(cache) if cache is not None else 0
            print(f"   col {c+1}/{n}: states={len(dist)} cache={csz} peakRSS={rss:.0f}MB", flush=True)
    return sum(dist.values())


# ---------------- independent brute force (tiny n) ----------------

def a_brute(n):
    if n == 0:
        return 1
    edges = []
    for r in range(n):
        for c in range(n):
            if c + 1 < n:
                edges.append(((r, c), (r, c + 1)))
            if r + 1 < n:
                edges.append(((r, c), (r + 1, c)))
            if r + 1 < n and c + 1 < n:
                edges.append(((r, c), (r + 1, c + 1)))
            if r + 1 < n and c - 1 >= 0:
                edges.append(((r, c), (r + 1, c - 1)))
    m = len(edges)
    matched = set()
    count = 0

    def dfs(i):
        nonlocal count
        if i == m:
            for u, w in edges:
                if u not in matched and w not in matched:
                    return
            count += 1
            return
        dfs(i + 1)                      # exclude edge i
        u, w = edges[i]
        if u not in matched and w not in matched:
            matched.add(u); matched.add(w)
            dfs(i + 1)                  # include
            matched.discard(u); matched.discard(w)

    dfs(0)
    return count


KNOWN = {1: 1, 2: 3, 3: 94, 4: 7148, 5: 2699317,
         6: 3897125868, 7: 25675258831176}

if __name__ == "__main__":
    args = sys.argv[1:]
    if args and args[0] == "brute":
        hi = int(args[1]) if len(args) > 1 else 4
        for n in range(1, hi + 1):
            print("brute", n, a_brute(n), "known", KNOWN.get(n), flush=True)
        sys.exit(0)
    verbose = "-v" in args
    nocache = "-nocache" in args
    args = [x for x in args if x not in ("-v", "-nocache")]
    lo = int(args[0]) if args else 1
    hi = int(args[1]) if len(args) > 1 else 8
    for n in range(lo, hi + 1):
        t0 = time.time()
        v = a_transfer(n, verbose=verbose, cache_trans=not nocache)
        dt = time.time() - t0
        rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024 * 1024)
        chk = KNOWN.get(n)
        tag = ("OK" if chk == v else ("MISMATCH exp=%d" % chk)) if chk is not None else "NEW"
        print(f"a({n}) = {v}   [{tag}]  {dt:.2f}s  peakRSS={rss:.0f}MB", flush=True)
