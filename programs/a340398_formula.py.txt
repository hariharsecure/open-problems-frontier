#!/usr/bin/env python3
"""
A340398: Number of spanning trees in the Bruhat graph (= transposition Cayley
graph) of the symmetric group S_n.

Vertices = permutations of {1..n} (n! of them). Edge between pi, sigma iff
pi*sigma^{-1} is a transposition (so the graph is (C(n,2))-regular).

This is a CAYLEY GRAPH on S_n with connection set = all transpositions, so its
Laplacian eigenvalues are indexed by irreducible representations (partitions
lambda of n): eigenvalue_lambda = C(n,2) - content_sum(lambda), with
multiplicity f_lambda^2 (f_lambda = dim of the irrep = # standard Young
tableaux of shape lambda, via the hook-length formula). The all-ones/trivial
partition (n) gives eigenvalue 0 (excluded). Kirchhoff (via Cayley-graph
eigenvalue product, Ehrenborg 2021):

  a(n) = (1/n!) * Product_{lambda |- n, lambda != (n)} eigenvalue_lambda ^ (f_lambda^2)
"""
from fractions import Fraction
from math import comb, factorial

def partitions(n, maxpart=None):
    if maxpart is None: maxpart = n
    if n == 0:
        yield ()
        return
    for k in range(min(n, maxpart), 0, -1):
        for rest in partitions(n-k, k):
            yield (k,) + rest

def hook_lengths_and_content(lam):
    # lam: partition as tuple of row lengths, descending.
    rows = len(lam)
    cols = [0]*rows
    conj = []  # conjugate partition
    maxc = lam[0] if rows else 0
    for c in range(maxc):
        cnt = sum(1 for r in lam if r > c)
        conj.append(cnt)
    hooks = []
    content_sum = 0
    for i in range(rows):          # 0-indexed row i (row i+1 in 1-index)
        for j in range(lam[i]):    # 0-indexed col j
            arm = lam[i] - j - 1
            leg = conj[j] - i - 1
            hooks.append(arm+leg+1)
            content_sum += (j - i)  # content (1-indexed j - i = 0-indexed j - i)
    return hooks, content_sum

def f_lambda(lam, n):
    hooks, _ = hook_lengths_and_content(lam)
    prod = 1
    for h in hooks: prod *= h
    return factorial(n) // prod

def bruhat_spanning_trees(n):
    if n == 1:
        return 1
    total = comb(n,2)
    result = Fraction(1, factorial(n))
    for lam in partitions(n):
        if lam == (n,):   # trivial partition -> eigenvalue 0, excluded
            continue
        hooks, csum = hook_lengths_and_content(lam)
        f = factorial(n)
        for h in hooks: f //= h
        eig = total - csum
        result *= Fraction(eig, 1) ** (f*f)
    assert result.denominator == 1, f"non-integer result for n={n}: {result}"
    return result.numerator

if __name__ == "__main__":
    known = {1:1, 2:1, 3:81, 4:22799473113563136}
    for n,want in known.items():
        got = bruhat_spanning_trees(n)
        print(f"n={n} got={got} want={want} match={got==want}")
        assert got == want
    for n in range(5, 9):
        v = bruhat_spanning_trees(n)
        print(f"NEW a({n}) = {v}")
