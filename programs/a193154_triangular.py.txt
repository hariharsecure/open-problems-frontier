#!/usr/bin/env python3
"""
A193154 — Number of spanning trees of the triangular graph T(n) (= Johnson
graph J(n,2) = line graph of K_n): vertices are the C(n,2) 2-subsets of
{1..n}; two vertices adjacent iff they share exactly one element.

T(n) is a strongly regular graph SRG(C(n,2), 2(n-2), n-2, 4) with KNOWN
adjacency eigenvalues:
  k = 2(n-2)                multiplicity 1
  n-4                       multiplicity n-1
  -2                        multiplicity n(n-3)/2
so Laplacian eigenvalues (k - adjacency eigenvalue):
  0                         multiplicity 1        (trivial, excluded)
  n                         multiplicity n-1
  2n-2                      multiplicity n(n-3)/2
By Kirchhoff: a(n) = (1/C(n,2)) * n^(n-1) * (2n-2)^(n(n-3)/2)   for n>=4
(n=2,3 are small/degenerate cases handled directly: T(2)=K_1 -> 1,
T(3)=K_3 -> 3).

Two independent methods here:
  Method A: explicit Johnson-graph Laplacian + Bareiss (no closed form used).
  Method B: the closed-form eigenvalue product above (pure integer arithmetic).
"""
import sys, time, itertools
from math import comb
from bareiss import bareiss_det

def triangular_laplacian(n):
    verts = list(itertools.combinations(range(n), 2))
    N = len(verts)
    idx = {v:i for i,v in enumerate(verts)}
    adj = [[] for _ in range(N)]
    for i,(a,b) in enumerate(verts):
        for j in range(i+1,N):
            c,d = verts[j]
            if len({a,b} & {c,d}) == 1:
                adj[i].append(j); adj[j].append(i)
    L = [[0]*(N-1) for _ in range(N-1)]
    for i in range(N-1):
        L[i][i] = len(adj[i])
        for j in adj[i]:
            if j < N-1: L[i][j] -= 1
    return L

def triangular_spanning_trees_explicit(n):
    if n <= 2: return 1
    return bareiss_det(triangular_laplacian(n))

def triangular_spanning_trees_formula(n):
    if n == 2: return 1
    if n == 3: return 3
    N = comb(n,2)
    m2 = n*(n-3)//2
    num = (n**(n-1)) * ((2*n-2)**m2)
    assert num % N == 0, f"non-integer at n={n}"
    return num // N

        if e is not None:
