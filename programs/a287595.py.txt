#!/usr/bin/env python3
"""
A287595: number of MAXIMAL matchings in the n x n grid graph.

A matching = set of edges, pairwise vertex-disjoint. MAXIMAL = no edge can be
added, i.e. there is NO edge whose both endpoints are unmatched. Equivalently:
the set of UNMATCHED vertices is an INDEPENDENT set (no two adjacent unmatched).

METHOD (exact, deterministic column transfer over big-ints; M24 pattern):
  Grid vertices (r,c), r,c in 0..n-1. Edges: horizontal (r,c)-(r,c+1) and
  vertical (r,c)-(r+1,c). Process column by column, left to right.

  Per-row boundary STATE between column c and c+1 takes 3 values, encoding how
  vertex (r,c) constrains column c+1:
     0 = "matched, no outgoing horizontal edge" (r,c) is matched; (r,c+1) free.
     1 = "H": horizontal edge (r,c)-(r,c+1) is present -> (r,c+1) is matched by it
         and cannot take any other edge.
     2 = "pending": (r,c) is UNMATCHED and no outgoing H edge -> for maximality
         (r,c+1) MUST be matched (otherwise edge (r,c)-(r,c+1) is addable).
  The full column boundary state is the tuple of these over all n rows.

  A vertex (r,c) is allowed to be unmatched only if it is NOT pending-forced and
  all its neighbours are matched:
     - left neighbour: incoming state must be 0 (matched). state 2 forces matched,
       state 1 means matched-by-H. So an unmatched vertex always has incoming 0.
       (Column 0 uses all-0 incoming; a phantom "matched" left neighbour, which
        correctly permits column-0 unmatched vertices.)
     - vertical neighbours: enforced within the column (no two adj. unmatched).
     - right neighbour: enforced by emitting outgoing state 2 (pending).

  Column transition (given incoming array a[0..n-1], choose vertical edges V within
  the column and outgoing right-edges R'):
     matched[r] = (a[r]==1) or ve[r-1] or ve[r] or R'[r]
     * a[r]==1  -> vertex already matched by incoming H: no V edge may touch r, R'[r]=0.
     * a[r]==2  -> must be matched: exactly one of {ve[r-1],ve[r],R'[r]} (>=1; matching gives <=1).
     * a[r]==0  -> free: matched or unmatched. If unmatched, emit pending.
     * V is a matching on the row-path (no two adjacent vertical edges).
     * R'[r]=1 requires vertex r otherwise unmatched-by-others (no incoming H, no V edge).
     * vertical maximality: rows r,r+1 may NOT both be unmatched.
     outgoing o[r] = 1 if R'[r] else (0 if matched[r] else 2)

  LAST column (c=n-1): force R'=0 (no column to the right). Any resulting outgoing
  state is acceptable: state 1 cannot occur (R'=0); state 2 means a last-column
  vertex is unmatched, which is fine (no right neighbour to force). So the answer
  is the sum of all state counts after the last column.

  Initial: column 0 incoming = all-0 (single state), count 1.
"""
import sys, time, resource

# ---------------- exact transfer-matrix engine ----------------
# State is a PACKED integer: 2 bits per row, digit(r) in {0,1,2} stored at bits [2r,2r+1].
#   0 = matched, no outgoing H          1 = outgoing horizontal edge (H)
#   2 = pending (unmatched -> forces next column's vertex matched)

def column_transitions(n, a_packed, last):
    """Given incoming PACKED state `a_packed` and whether this is the LAST column,
    enumerate all valid (vertical-edges, right-edges) fillings and return a dict
    {packed_outgoing_state: multiplicity}.

    Inner row DP carries:
       prev_ve : vertical edge placed from row r-1 into row r (occupies r's top)
       prev_un : row r-1 was left UNMATCHED (forbid two vertically-adjacent unmatched)
    """
    a = [(a_packed >> (2 * r)) & 3 for r in range(n)]
    out = {}
    ob = [0] * n  # outgoing digit per row

    def rec(r, prev_ve, prev_un):
        if r == n:
            key = 0
            for i in range(n - 1, -1, -1):
                key = (key << 2) | ob[i]
            out[key] = out.get(key, 0) + 1
            return
        ar = a[r]
        for ve_down in (0, 1):
            if ve_down and r == n - 1:
                continue                 # no row below
            if ve_down and prev_ve:
                continue                 # would share vertex r
            for Rp in (0, 1):
                if last and Rp:
                    continue             # no right edges out of last column
                if ar == 1:
                    # locked by incoming H: no other incident edge allowed
                    if prev_ve or ve_down or Rp:
                        continue
                    matched = True
                else:
                    deg = (1 if prev_ve else 0) + ve_down + Rp
                    if deg > 1:
                        continue
                    matched = deg == 1
                if ar == 2 and not matched:
                    continue             # pending vertex must be matched
                un = not matched
                if un and ar != 0:
                    continue             # unmatched needs left neighbour matched (ar==0)
                if un and prev_un:
                    continue             # vertical maximality
                ob[r] = 1 if Rp else (0 if matched else 2)
                rec(r + 1, ve_down, un)
    rec(0, 0, False)
    return out


def a_transfer(n, verbose=False, cache_trans=True):
    if n == 0:
        return 1
    if n == 1:
        return 1
    dist = {0: 1}                 # packed all-zero start state
    cache = {} if cache_trans else None  # packed_in -> dict(packed_out -> mult) [non-last cols]
    for c in range(n):
        last = (c == n - 1)
        nd = {}
        ndget = nd.get
        if last:
            for st, cnt in dist.items():
                for o, m in column_transitions(n, st, last).items():
                    nd[o] = ndget(o, 0) + cnt * m
        elif cache is not None:
            for st, cnt in dist.items():
                tr = cache.get(st)
                if tr is None:
                    tr = column_transitions(n, st, False)
                    cache[st] = tr
                for o, m in tr.items():
                    nd[o] = ndget(o, 0) + cnt * m
        else:
            for st, cnt in dist.items():
                for o, m in column_transitions(n, st, False).items():
                    nd[o] = ndget(o, 0) + cnt * m
        dist = nd
        if verbose:
            rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024*1024)
            csz = len(cache) if cache is not None else 0
            print(f"   col {c+1}/{n}: states={len(dist)} cache={csz} peakRSS={rss:.0f}MB", flush=True)
    return sum(dist.values())


# ---------------- CSR transfer engine (memory-light, for large n) ----------------
# Build the transfer operator ONCE as compact arrays (the reachable-state set is
# closed and small), then apply it n-1 times. Avoids the giant dict-of-dicts cache
# AND per-column recomputation.

def a_transfer_csr(n, verbose=False):
    import array
    if n == 0 or n == 1:
        return 1
    # 1. BFS-discover reachable states from the all-zero start, following NON-LAST
    #    transitions, building the compact CSR arrays DIRECTLY (no bulky temporary).
    # NOTE: every (incoming, outgoing) transition has multiplicity EXACTLY 1 -- given the
    # incoming state and the outgoing state, the vertical-edge set and outgoing horizontal
    # edges are uniquely determined (verified: max mult == 1 for all n tested). So we store
    # NO data array; a repeated append here would still be correct if that ever changed.
    idx = {0: 0}          # packed_state -> index
    order = [0]           # index -> packed_state
    indptr = array.array('Q'); indptr.append(0)
    indices = array.array('I')
    head = 0
    while head < len(order):
        s = order[head]
        for o, m in column_transitions(n, s, False).items():
            j = idx.get(o)
            if j is None:
                j = len(order); idx[o] = j; order.append(o)
            for _ in range(m):        # m is always 1; loop keeps it correct if not
                indices.append(j)
        indptr.append(len(indices))
        head += 1
    ns = len(order)
    nnz = len(indices)
    if verbose:
        rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024*1024)
        print(f"   n={n}: reachable states={ns} nnz={nnz}  CSR built  peakRSS={rss:.0f}MB", flush=True)
    # 3. Last-column weight L[s] = total number of valid last-column fillings from s.
    L = array.array('Q')
    for s in order:
        tot = 0
        for m in column_transitions(n, s, True).values():
            tot += m
        L.append(tot)
    if verbose:
        rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024*1024)
        print(f"   n={n}: L built  peakRSS={rss:.0f}MB", flush=True)
    # 4. DP: start vector e_0; apply non-last transfer (n-1) times; then contract with L.
    v = [0] * ns
    v[0] = 1
    for c in range(n - 1):
        nv = [0] * ns
        for i in range(ns):
            vi = v[i]
            if vi:
                lo = indptr[i]; hi = indptr[i + 1]
                for k in range(lo, hi):
                    nv[indices[k]] += vi
        v = nv
        if verbose:
            rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024*1024)
            print(f"   n={n}: applied col {c+1}/{n-1}  peakRSS={rss:.0f}MB", flush=True)
    total = 0
    for i in range(ns):
        if v[i]:
            total += v[i] * L[i]
    return total


# ---------------- independent brute force (tiny n) ----------------

def a_brute(n):
    """All matchings of the n x n grid via edge enumeration; count maximal ones."""
    if n == 0:
        return 1
    # edges
    edges = []
    for r in range(n):
        for c in range(n):
            if c + 1 < n:
                edges.append(((r, c), (r, c + 1)))
            if r + 1 < n:
                edges.append(((r, c), (r + 1, c)))
    V = [(r, c) for r in range(n) for c in range(n)]
    # adjacency for maximality check
    adj = {v: set() for v in V}
    for u, w in edges:
        adj[u].add(w); adj[w].add(u)
    m = len(edges)
    count = 0
    # enumerate matchings by DFS over edges (include/exclude), pruning conflicts
    used = set()
    matched = set()

    def dfs(i):
        nonlocal count
        if i == m:
            # maximality: no edge with both endpoints unmatched
            for u, w in edges:
                if u not in matched and w not in matched:
                    return
            count += 1
            return
        u, w = edges[i]
        # exclude
        dfs(i + 1)
        # include if both free
        if u not in matched and w not in matched:
            matched.add(u); matched.add(w)
            dfs(i + 1)
            matched.discard(u); matched.discard(w)
    dfs(0)
    return count


KNOWN = {0:1,1:1,2:2,3:22,4:400,5:22228,6:3136370,7:1158560776,
         8:1147204164108,9:2980178704765860,10:20513821200001569410,
         11:373243563814532182524614,12:17941038966060235808302667164}

if __name__ == "__main__":
    args = sys.argv[1:]
    if args and args[0] == "brute":
        hi = int(args[1]) if len(args) > 1 else 4
        for n in range(0, hi + 1):
            print("brute", n, a_brute(n), "known", KNOWN.get(n), flush=True)
        sys.exit(0)
    verbose = "-v" in args
    nocache = "-nocache" in args
    dicteng = "-dict" in args    # use dict engine instead of CSR
    args = [x for x in args if x not in ("-v", "-nocache", "-dict")]
    lo = int(args[0]) if args else 0
    hi = int(args[1]) if len(args) > 1 else 12
    for n in range(lo, hi + 1):
        t0 = time.time()
        if dicteng:
            v = a_transfer(n, verbose=verbose, cache_trans=not nocache)
        else:
            v = a_transfer_csr(n, verbose=verbose)
        dt = time.time() - t0
        rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / (1024*1024)  # MB (bytes on macOS)
        chk = KNOWN.get(n)
        tag = ("OK" if chk == v else ("MISMATCH exp=%d" % chk)) if chk is not None else "NEW"
        print(f"a({n}) = {v}   [{tag}]  {dt:.2f}s  peakRSS={rss:.0f}MB", flush=True)
