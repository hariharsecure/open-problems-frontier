#!/usr/bin/env python3
"""
A378860: number of MINIMAL edge covers of the n-Sierpinski gasket graph SG_n.

SG_1 = triangle K_3. SG_n = 3 copies of SG_{n-1} glued pairwise at their
3 corner vertices (top / bottom-left / bottom-right). |E(SG_n)| = 3^n,
|V(SG_n)| = (3^n + 3)/2. Each corner vertex has degree 2 in the full graph.

CHARACTERIZATION (same as M30, standard): a minimal edge cover = a spanning
subgraph H with (A) every vertex degree >= 1 and (B) every edge has a
degree-1 endpoint  <=>  every connected component of H is a star K_{1,k}.
Count SUBGRAPHS H satisfying (A)+(B).

RENORMALIZATION (Lineage 1):
Track the exact count of partial configurations as a function of the 3 corner
vertices' BOUNDARY STATE s=(c,u):
    c in {0,1,2} : local degree of the corner (# chosen edges incident to it
                   inside this copy)
    u in {0,..,c}: # of those incident edges whose OTHER endpoint (interior to
                   the copy) has degree >= 2  -> that edge's (B) is NOT yet
                   satisfied and is DEFERRED to the corner becoming degree 1.
All interior (non-corner) vertices and all edges not incident to a corner have
their (A)/(B) constraints enforced immediately; only the corners' own (A) and
their incident edges' (B) are deferred into the state.

f_n : (s_top, s_left, s_right) -> count.

Base f_2 built by direct brute force over the 2^9 subsets of the explicit SG_2
graph (6 vertices, 9 edges; corners A,B,C degree 2, midpoints X,Y,Z interior).

Gluing SG_n from copies P(top),Q(left),R(right):
    g1 = P.l & Q.t ;  g2 = P.r & R.t ;  g3 = Q.r & R.l   (interior after merge)
    exposed corners of SG_n = (P.t, Q.l, R.r).
A merge of two corner states (ca,ua),(cb,ub) is legal iff
    D = ca+cb >= 1  AND  (ua+ub == 0  OR  D == 1).
Closing an exposed corner (top level): c>=1 AND (u==0 OR c==1).
"""
import sys, time

# ---------- base f_2 by explicit brute force ----------

def build_f2():
    # vertices: corners A,B,C ; midpoints X,Y,Z (interior)
    A,B,C,X,Y,Z = range(6)
    corners = (A,B,C)                       # top, bottom-left, bottom-right
    # edges of SG_2 (three small triangles)
    edges = [
        (A,X),(A,Y),(X,Y),      # top triangle
        (X,B),(X,Z),(B,Z),      # bottom-left triangle
        (Y,Z),(Y,C),(Z,C),      # bottom-right triangle
    ]
    m = len(edges)
    f2 = {}
    for mask in range(1 << m):
        deg = [0]*6
        chosen = []
        for i in range(m):
            if mask >> i & 1:
                u,v = edges[i]
                deg[u]+=1; deg[v]+=1
                chosen.append((u,v))
        # interior vertices must satisfy (A): degree>=1
        if deg[X]==0 or deg[Y]==0 or deg[Z]==0:
            continue
        ok = True
        for (u,v) in chosen:
            u_corner = u in corners
            v_corner = v in corners
            if not u_corner and not v_corner:
                # interior-interior edge: enforce (B) now
                if deg[u]!=1 and deg[v]!=1:
                    ok=False; break
            # corner-incident edge: (B) deferred to the corner state (u-count)
        if not ok:
            continue
        # corner states
        st = []
        good = True
        for c in corners:
            cc = deg[c]                       # 0,1,2
            uu = 0
            for (u,v) in chosen:
                if u==c or v==c:
                    other = v if u==c else u
                    if deg[other] >= 2:        # far endpoint degree>=2 -> deferred
                        uu += 1
            st.append((cc,uu))
        key = (st[0],st[1],st[2])
        f2[key] = f2.get(key,0)+1
    return f2

# ---------- merge / close rules ----------

def merge_ok(sa, sb):
    ca,ua = sa; cb,ub = sb
    D = ca+cb
    if D < 1: return False
    return (ua+ub == 0) or (D == 1)

def closable(s):
    c,u = s
    return c >= 1 and (u == 0 or c == 1)

# ---------- recursion f_n from f_{n-1} ----------

def step(fprev):
    # group prev by each corner for efficiency, but 6^3 small -> brute triples.
    states = list(fprev.keys())
    fn = {}
    # precompute list once
    for (Pt,Pl,Pr) in states:
        cP = fprev[(Pt,Pl,Pr)]
        for (Qt,Ql,Qr) in states:
            if not merge_ok(Pl,Qt):   # g1
                continue
            cPQ = cP*fprev[(Qt,Ql,Qr)]
            for (Rt,Rl,Rr) in states:
                if not merge_ok(Pr,Rt):   # g2
                    continue
                if not merge_ok(Qr,Rl):   # g3
                    continue
                key = (Pt,Ql,Rr)
                fn[key] = fn.get(key,0) + cPQ*fprev[(Rt,Rl,Rr)]
    return fn

def close(f):
    tot = 0
    for (s1,s2,s3),cnt in f.items():
        if closable(s1) and closable(s2) and closable(s3):
            tot += cnt
    return tot

# ---------- explicit-graph brute (ground-truth anchor, small n) ----------

def sg_graph(n):
    """Return (vertices set, edges list) of SG_n with the 3 corners.
    Returns (edges, corners) where corners=(top,left,right) as vertex ids."""
    # represent vertices by frozenset-free integer ids via recursive build using
    # coordinate tuples; corners are geometric extreme points.
    if n==1:
        # triangle corners 0,1,2
        edges=[(0,1),(1,2),(0,2)]
        return edges,(0,1,2)
    esub,(t,l,r)=sg_graph(n-1)
    # three copies P,Q,R with vertex namespacing (copy,vertex)
    def relabel(copy, vid):
        return (copy,vid)
    edges=[]
    for copy in range(3):
        for (a,b) in esub:
            edges.append((relabel(copy,a),relabel(copy,b)))
    # gluing: unify vertices via union-find
    parent={}
    def find(x):
        parent.setdefault(x,x)
        while parent[x]!=x:
            parent[x]=parent[parent[x]]; x=parent[x]
        return x
    def union(a,b):
        ra,rb=find(a),find(b)
        if ra!=rb: parent[ra]=rb
    P,Q,R=0,1,2
    # P=top, Q=left, R=right ; corners (t,l,r)
    union(relabel(P,l),relabel(Q,t))  # g1
    union(relabel(P,r),relabel(R,t))  # g2
    union(relabel(Q,r),relabel(R,l))  # g3
    # remap to ints
    edges2=[]
    for (a,b) in edges:
        edges2.append((find(a),find(b)))
    corners=(find(relabel(P,t)),find(relabel(Q,l)),find(relabel(R,r)))
    # normalize vertex ids
    verts=set()
    for (a,b) in edges2: verts.add(a); verts.add(b)
    idx={v:i for i,v in enumerate(sorted(verts,key=str))}
    edges3=[(idx[a],idx[b]) for (a,b) in edges2]
    corners3=tuple(idx[c] for c in corners)
    return edges3,corners3

def brute(n):
    edges,corners=sg_graph(n)
    verts=set()
    for a,b in edges: verts.add(a); verts.add(b)
    V=len(verts); m=len(edges)
    count=0
    for mask in range(1<<m):
        deg=[0]*V
        chosen=[]
        for i in range(m):
            if mask>>i&1:
                a,b=edges[i]; deg[a]+=1; deg[b]+=1; chosen.append((a,b))
        if any(d==0 for d in deg): continue
        ok=True
        for a,b in chosen:
            if deg[a]!=1 and deg[b]!=1:
                ok=False; break
        if ok: count+=1
    return count

KNOWN={1:3,2:17,3:9463,4:1822981173460}

def run(hi):
    # a(1) via triangle brute
    res={}
    res[1]=brute(1)
    f=build_f2()
    res[2]=close(f)
    for n in range(3,hi+1):
        f=step(f)
        res[n]=close(f)
    return res

if __name__=="__main__":
    args=sys.argv[1:]
    if args and args[0]=="brute":
        hi=int(args[1]) if len(args)>1 else 3
        for n in range(1,hi+1):
            t0=time.time(); v=brute(n); print(f"brute a({n})={v} known={KNOWN.get(n)} {time.time()-t0:.2f}s",flush=True)
        sys.exit(0)
    hi=int(args[0]) if args else 6
    t0=time.time()
    res=run(hi)
    for n in sorted(res):
        chk=KNOWN.get(n)
        tag="OK" if chk==res[n] else (f"MISMATCH exp={chk}" if chk is not None else "NEW")
        print(f"a({n}) = {res[n]}   [{tag}]",flush=True)
    print(f"total {time.time()-t0:.3f}s",flush=True)
